第 1 章 ROS2介绍与安装
# 安装ros2 
```
sudo apt update 
sudo apt upgrade 

# 检测网络是否联通
ping -w 5  fishros.com 
wget http://fishros.com/install -O fishros && . fishros


# 验证是否安装成功， 打开新终端
ros2 -h 
ros2 run --help

# 一键安装后各个依赖的版本
gcc (Ubuntu 11.4.0-1ubuntu1~22.04) 11.4.0
g++ (Ubuntu 11.4.0-1ubuntu1~22.04) 11.4.0
cmake version 3.22.1
Python 3.10.12 # python3 --version 不是Python2 

sqlite 没有  
make 没有需要安装 
 sudo apt install make

rqt 依赖的库
 Python 3.10.12, PyQt 5.15.6 (QtCore, QtDBus, QtDesigner, QtGui, QtHelp, QtNetwork, QtPrintSupport, QtSvg, QtTest, QtWidgets, QtXml), Qt 5.15.3, SIP C++ bindings available



# 怎么查看ros2的版本 
echo $ROS_DISTRO
humble

# 运行第一个程序
ros2 run turtlesim turtlesim_node

# 使用按键控制海龟
ros2 run turtlesim turtle_teleop_key
```
